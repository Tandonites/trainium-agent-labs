    input push,
    input pop,
    input [7:0] data_in,
    output reg [7:0] data_out,

    output reg empty,
    output reg [2:0] count
);

    // FIFO parameters
    localparam DEPTH = 4;
    localparam ADDR_WIDTH = 2;

    // Internal signals
    reg [ADDR_WIDTH-1:0] write_ptr, read_ptr;
    reg [ADDR_WIDTH-1:0] next_write_ptr, next_read_ptr;
    reg [7:0] mem [0:DEPTH-1];

    // State variables
    reg [2:0] next_count;

    // Compute next pointers and count
    always @(posedge clk) begin
        if (reset) begin
            write_ptr <= 0;
            read_ptr <= 0;
            next_write_ptr <= 0;
            next_read_ptr <= 0;
            next_count <= 0;
        end else begin
            // Compute next pointers
            next_write_ptr <= write_ptr + 1;
            next_read_ptr <= read_ptr + 1;

            // Compute next count
            if (push && pop) begin
                // Both push and pop are accepted
                next_count <= count;
            end else if (push) begin
                // Only push is accepted
                if (count < DEPTH) begin
                    next_count <= count + 1;
                end else begin
                    next_count <= count;
                end
            end else if (pop) begin
                // Only pop is accepted
                if (count > 0) begin
                    next_count <= count - 1;
                end else begin
                    next_count <= count;
                end

                // No operation
                next_count <= count;
            end

            // Update pointers and count
            write_ptr <= next_write_ptr;
            read_ptr <= next_read_ptr;
            count <= next_count;
        end

            full <= 0;
            empty <= 1;
