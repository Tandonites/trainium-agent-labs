module binary_to_bcd_converter(input [4:0] binary_input,
                                output [7:0] bcd_output);

sequence. Otherwise it is 0. Fewer than eight sampled values cannot match.
Detection is a registered one-cycle pulse, not a sticky flag. Overlapping
matches are allowed; a trailing value can also begin a later match.

    input clk,
    input reset_n,
    input [2:0] data,
    output reg sequence_found
);

    // Define the target sequence
    localparam [2:0] seq[7] = '{3'b001, 3'b101, 3'b110, 3'b000,
                               3'b110, 3'b110, 3'b011, 3'b101};

    // State machine to track the current position in the sequence
    reg [2:0] current_state;

    // Shift register to hold the last 8 data samples
    reg [23:0] data_shift_reg;

    // Internal signal to indicate if the current data matches the expected value
    reg match;

    // Always block for clocked logic
    always @(posedge clk) begin
        if (!reset_n) begin
            // Reset: clear shift register and state
            data_shift_reg <= 24'b0;
            current_state <= 3'b000;
            sequence_found <= 1'b0;
        end else begin
            // Shift in new data
            data_shift_reg <= {data_shift_reg[18:0], data};

            // Check if current data matches the expected value for the current state
            match = (data_shift_reg[23:20] == seq[current_state]);

            // Update state if match is found
            if (match) begin
                current_state <= current_state + 1;
                if (current_state == 3'b111) begin
                    // Full sequence matched
                    sequence_found <= 1'b1;
                    current_state <= 3'b000;
                end
            end else begin
                // No match, reset state if we've gone too far
                if (current_state != 3'b000) begin
                    current_state <= 3'b000;
                end
            end

            // Clear sequence_found after one cycle
            if (current_state == 3'b000 && data_shift_reg[23:20] == seq[0]) begin
                sequence_found <= 1'b1;
            end else if (current_state != 3'b000) begin
                sequence_found <= 1'b0;
            end
        end
    end

endmodule
