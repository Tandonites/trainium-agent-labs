    output reg [2:0] count

    // FIFO parameters
    localparam DEPTH = 4;
    localparam ADDR_WIDTH = 2;

    // Internal signals

            next_read_ptr <= 0;
            next_count <= 0;

            next_read_ptr <= read_ptr + 1;

                end else begin
                    next_count <= count;

                // Only pop is accepted

                    next_count <= count - 1;
                end else begin
                    next_count <= count;
                end

            end

            read_ptr <= next_read_ptr;
            count <= next_count;
